i < 5; i++) { ctx.save() ctx.rotate((Math.PI * 2 * i) / 5) ctx.beginPath() ctx.moveTo(0, 0) ctx.bezierCurveTo(-38, -10, -45, -48, 0, -62) ctx.bezierCurveTo(45, -48, 38, -10, 0, 0) ctx.fillStyle = "#f29ab5" ctx.fill() ctx.restore() } ctx.beginPath() ctx.arc(0, 0, 14, 0, Math.PI * 2) ctx.fillStyle = "#171017" ctx.fill() ctx.restore() const slimeTop = centerY + 32 const drips = [ { x: centerX - 23, width: 12, length: 42, delay: 0, }, { x: centerX, width: 18, length: 82, delay: 1.2, }, { x: centerX + 24, width: 11, length: 52, delay: 2.4, }, ] drips.forEach((drip) => { const movement = Math.sin(time * 0.002 + drip.delay) * 6 const length = drip.length + movement ctx.beginPath() ctx.moveTo( drip.x - drip.width / 2, slimeTop ) ctx.lineTo( drip.x - drip.width / 2, slimeTop + length ) ctx.quadraticCurveTo( drip.x, slimeTop + length + 12, drip.x + drip.width / 2, slimeTop + length ) ctx.lineTo( drip.x + drip.width / 2, slimeTop ) ctx.closePath() ctx.fillStyle = "#8f174f" ctx.fill() }) const dropCycle = (time * 0.035) % 110 if (dropCycle > 35 && dropCycle < 100) { ctx.beginPath() ctx.arc( centerX, slimeTop + dropCycle, 4, 0, Math.PI * 2 ) ctx.fillStyle = "#8f174f" ctx.fill() } time += 16 animationFrame = requestAnimationFrame(draw) } draw() return () => { window.removeEventListener("resize", resize) cancelAnimationFrame(animationFrame) } }, []) return (
